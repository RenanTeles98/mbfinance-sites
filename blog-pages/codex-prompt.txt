=== admin-core.js (resumo) ===

function init() {
  const authStatus = localStorage.getItem('mb_admin_auth');
  if (!authStatus) {
    document.getElementById('admin-app').style.display = 'none';
    return;
  }
  document.getElementById('admin-app').style.display = 'flex';
  if (typeof loadPosts === 'function') loadPosts();
  if (typeof renderAnalytics === 'function') renderAnalytics();
  if (typeof updateOfficialBlogUi === 'function') updateOfficialBlogUi();
}
document.addEventListener('DOMContentLoaded', init);

=== app/admin/route.ts ===

export async function GET() {
  const html = fs.readFileSync(path.join(process.cwd(), 'private', 'blog-admin.html'), 'utf-8');
  return new NextResponse(html, {
    headers: {
      'Content-Type': 'text/html; charset=utf-8',
      'Cache-Control': 'no-store, no-cache, must-revalidate',
    },
  });
}
